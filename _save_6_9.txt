            await new Promise(r => setTimeout(r, 2000));
        } catch(e) {
            results.push({index: art.index, title: art.title, code: -1, message: 'JS_ERROR: ' + e.message, pgc_id: null, img_count: art.img_count});
        }
    }
    return JSON.stringify(results);
})();
